        System.out.println("정렬 전 : " + Arrays.toString(arr));
        // 선택 정렬 수행
        for(int i=0;i<N;i++) {
            int idx = i;
            int least = arr[i];
            // System.out.println("\ni:" + i + "\nj:");
            for(int j=i;j<N;j++) {
                // System.out.print(j + " ");
                if(arr[j] < least) {
                    least = arr[j];
                    idx = j;
                }
            }
            int temp = arr[i];
            arr[i] = least;
            arr[idx] = temp;
        }
        System.out.println("정렬 후 : " + Arrays.toString(arr));
